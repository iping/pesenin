const algoCode = `// Called inside the CREATE ORDER transaction (SERIALIZABLE or with an advisory lock per bank account)
func assignUniqueAmount(ctx, tx, bankAccountID, base int64) (suffix, total int64, err error) {
    // 1. Load taken totals for this bank account among active orders
    taken := tx.TakenTotals(ctx, bankAccountID) // payment_status IN ('pending','proof_submitted')
                                                // OR (expired AND expired_at > now() - 48h)
    // 2. Try random suffixes from 1..999, skipping ones that make "round-looking" numbers
    for _, s := range shuffled(1..999) {
        if s%100 == 0 { continue }            // avoid 45.100, 45.200
        if taken.Has(base + s) { continue }
        return s, base + s, nil
    }
    // 3. Extremely rare: widen to 1.001..1.999, then return ErrNoSuffix and ask seller to add a bank account
}
// Backed by: CREATE UNIQUE INDEX ux_orders_active_amount ON orders(bank_account_id, total_due)
//            WHERE payment_status IN ('pending','proof_submitted');`;

            total_qty                if c.apply_level == per_item
            qty_of(c.products)       if c.apply_level == per_product
    amt = c.amount_rp * basis                                    if c.amount_type == fixed
          ceil(discounted_or_product_subtotal * c.percent_bps / 10000) if c.amount_type == percent
    amt = min(amt, c.max_amount_rp) if c.max_amount_rp
    if c.waive_min_subtotal_rp and discounted >= c.waive_min_subtotal_rp: amt = 0, waived

base_total = discounted + sum(amt)
total_due  = base_total + unique_suffix
